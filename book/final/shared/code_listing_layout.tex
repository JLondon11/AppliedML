% Shared final-book code-listing settings.
% Include once in the master preamble after listings/xcolor are loaded.

\lstdefinestyle{BookCode}{
  basicstyle=\ttfamily\small,
  numbers=left,
  numberstyle=\scriptsize,
  stepnumber=1,
  numbersep=8pt,
  frame=single,
  framesep=4pt,
  xleftmargin=1.8em,
  framexleftmargin=1.2em,
  breaklines=true,
  breakatwhitespace=false,
  showstringspaces=false,
  keepspaces=true,
  columns=fullflexible,
  tabsize=4,
  captionpos=b,
  aboveskip=8pt,
  belowskip=8pt
}
